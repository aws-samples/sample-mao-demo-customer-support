  - To find the products a specific customer purchased (e.g., by product_name), you MUST JOIN purchase_history to product_catalog ON product_id, filter customer_id on purchase_history, and filter product_name on product_catalog. <athena_schemas>
<product_catalog_table_schema>
CREATE EXTERNAL TABLE IF NOT EXISTS mac_prod_rec.product_catalog (
    product_id: STRING,
    product_name: STRING,
    category: STRING,
    price: DOUBLE,
    description: STRING,
    rating: DOUBLE,
    popularity: STRING
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
WITH SERDEPROPERTIES (
    'serialization.format' = ','
) LOCATION 's3://mac-demo-data-{account_number}/'
TBLPROPERTIES ('skip.header.line.count'='1');
</product_catalog_table_schema>

<purchase_history_table_schema>
CREATE EXTERNAL TABLE IF NOT EXISTS mac_prod_rec.purchase_history (
    customer_id: STRING,
    product_id: STRING,
    purchase_date: STRING,
    quantity: INT,
    purchase_amount: DOUBLE,
    payment_method: STRING
)
ROW FORMAT SERDE 'org.apache.hadoop.hive.serde2.OpenCSVSerde'
WITH SERDEPROPERTIES (
    'serialization.format' = ','
) LOCATION 's3://mac-demo-data-{account_number}/'
TBLPROPERTIES ('skip.header.line.count'='1');
</purchase_history_table_schema>

</athena_schemas>

<!-- Products a specific customer purchased, filtered by product_name.
     product_name lives on product_catalog and customer_id lives on
     purchase_history, so join the two on product_id — never filter
     product_catalog by customer_id (that column does not exist there). -->
<athena_example>
SELECT ph.customer_id, pc.product_name, pc.category, pc.price, ph.purchase_date, ph.quantity
FROM mac_prod_rec.purchase_history AS ph
JOIN mac_prod_rec.product_catalog AS pc
  ON ph.product_id = pc.product_id
WHERE ph.customer_id = 'cust005'
  AND pc.product_name LIKE '%promax%';
</athena_example>

<!-- Recommend catalog products in the categories a customer has purchased. -->
<athena_example>
SELECT DISTINCT pc.product_name, pc.category, pc.price, pc.rating
FROM mac_prod_rec.product_catalog AS pc
WHERE pc.category IN (
    SELECT DISTINCT pc2.category
    FROM mac_prod_rec.purchase_history AS ph
    JOIN mac_prod_rec.product_catalog AS pc2
      ON ph.product_id = pc2.product_id
    WHERE ph.customer_id = 'cust005'
)
ORDER BY pc.rating DESC
LIMIT 10;
</athena_example>
</athena_examples>
